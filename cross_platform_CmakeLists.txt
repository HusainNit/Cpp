cmake_minimum_required(VERSION 3.28)
project(secondApp)

# --- 1. Universal Settings (Works everywhere) ---
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# MSVC Runtime logic - CMake ignores this on Linux/Mac automatically
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# --- 2. Dependencies ---
include(FetchContent)
FetchContent_Declare(
    spdlog
    GIT_REPOSITORY https://github.com/gabime/spdlog.git
    GIT_TAG v1.17.0
)
FetchContent_MakeAvailable(spdlog)

# --- 3. Sources ---
file(GLOB_RECURSE MY_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_executable(${PROJECT_NAME} "${MY_SOURCES}")

# --- 4. Includes & Links ---
target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include/")
target_link_libraries(${PROJECT_NAME} PRIVATE spdlog::spdlog)

# PCH works on most modern compilers (MSVC, GCC, Clang)
target_precompile_headers(${PROJECT_NAME} PRIVATE 
    <vector> <string> <memory> <spdlog/spdlog.h>
)

# --- 5. Compiler-Specific Flags (The "Cross-Platform" Magic) ---

if(MSVC)
    # Windows / MSVC Flags
    target_compile_options(${PROJECT_NAME} PRIVATE 
        /W4 /WX /sdl /permissive- /utf-8 /MP /arch:AVX2 /Zc:inline
        $<$<CONFIG:Debug>:/Od>
        $<$<CONFIG:Release>:/fp:fast>
    )
    target_compile_definitions(${PROJECT_NAME} PUBLIC _CRT_SECURE_NO_WARNINGS)
    
    # GUI App logic (Windows only)
    set_target_properties(${PROJECT_NAME} PROPERTIES 
        LINK_FLAGS_DEBUG "/INCREMENTAL /DEBUG:FASTLINK"
        LINK_FLAGS "/SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup"
    )
else()
    # Linux & macOS (GCC or Clang)
    target_compile_options(${PROJECT_NAME} PRIVATE 
        -Wall -Wextra -Werror    # Equivalent to /W4 /WX
        -march=native            # Best optimization for the current CPU
        $<$<CONFIG:Debug>:-O0 -g> # Debug: No opt, include symbols
        $<$<CONFIG:Release>:-O3>  # Release: Full optimization
    )
endif()

# --- 6. IDE Organization ---
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/src" PREFIX "Source Files" FILES ${MY_SOURCES})
