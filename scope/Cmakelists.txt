cmake_minimum_required(VERSION 3.28)

# this to wake the compiler
project(scope VERSION 1.0.0 LANGUAGES CXX)

# make the app portable to other (add c to .exe so it run every where)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(CMAKE_CXX_STANDARD 20) # stander C++20 use
set(CMAKE_CXX_STANDARD_REQUIRED ON) # make sure to use C++20, and fail other wise
set(CMAKE_CXX_EXTENSIONS OFF) # use c++ standerd terms and prevent compiler shortcats

# Route all compiled binaries to a clean /bin/ folder
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Automatic file list
file(GLOB_RECURSE MY_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")

file(GLOB_RECURSE MY_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp")

# add the files to the IDA looks, use the glope in top - ${PROJECT_NAME} can be better sometimes
add_executable(${PROJECT_NAME} ${MY_SOURCES} ${MY_HEADERS})

# Include directories (Internal src + Public include)
target_include_directories(${PROJECT_NAME}
    PRIVATE 
        ${CMAKE_CURRENT_SOURCE_DIR}/src
    PUBLIC 
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

# Organizing the Files Tree structure
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/src" PREFIX "Source Files" FILES ${MY_SOURCES})
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/include" PREFIX "Header Files" FILES ${MY_HEADERS})

# compiler to compile heavy headers one time in bolb if not changed - ${PROJECT_NAME} can be better sometimes
target_precompile_headers(${PROJECT_NAME} PRIVATE
    <vector>
    <string>
    <memory>
)

# Enable Link-Time Optimization (IPO/LTO) for Release builds
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_supported OUTPUT error)
if(ipo_supported)
    set_property(TARGET ${PROJECT_NAME} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    message(STATUS "IPO/LTO enabled for Release builds")
endif()

# Optional: Speeds up compilation for large projects by batching source files
set_target_properties(${PROJECT_NAME} PROPERTIES UNITY_BUILD ON)

# Enable testing framework support
enable_testing()

if(MSVC)  # If using the VS compiler...

    #make sure SIMD optimizations take place (CPUs after 2013 only will run the .exe)
    add_compile_options(/arch:AVX2) 

    # 1. Rules for YOUR code
    target_compile_options(${PROJECT_NAME} PRIVATE 
        /W4      # Warning Level 4
        /WX      # Treat warnings as errors
        /sdl     # Enable Security Development Lifecycle checks
    )

    # 2. Rules for THIRD PARTY (External) code
    target_compile_options(${PROJECT_NAME} PRIVATE 
        /external:W3            # Treat external headers as Level 3
        /external:anglebrackets # Treat everything in < > as external
    )

    # make it multy processer compile
    target_compile_options(${PROJECT_NAME} PRIVATE /MP)

    # Enable faster Debug builds and better "Edit and Continue"
    target_compile_options(${PROJECT_NAME} PRIVATE /Zi) 

    # Better Debug Info Format
    target_compile_options(${PROJECT_NAME} PRIVATE /Zc:inline)
    
    # Improve the "Auto" window in VS - helps the debugger see variables better
    target_compile_options(${PROJECT_NAME} PRIVATE $<$<CONFIG:Debug>:/Od>) # Only for Debug

    target_compile_options(${PROJECT_NAME} PRIVATE /analyze)

    # Enable "Standards Conformance"
    target_compile_options(${PROJECT_NAME} PRIVATE /permissive-)

    # define a preprocessor macro when building your target.
	target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC _CRT_SECURE_NO_WARNINGS)

    # The "UTF-8" Fix
    target_compile_options(${PROJECT_NAME} PRIVATE /utf-8)

    # Enable AddressSanitizer for Debug builds to catch memory bugs instantly
    target_compile_options(${PROJECT_NAME} PRIVATE $<$<CONFIG:Debug>:/fsanitize=address>)

    # Enable Incremental Linking for faster "Edit and Continue"  - INCREMENTAL
    # Build Speed: "Fast Link"  - DEBUG:FASTLINK
    set_target_properties(${PROJECT_NAME} PROPERTIES LINK_FLAGS_DEBUG "/INCREMENTAL /DEBUG:FASTLINK")

endif()
