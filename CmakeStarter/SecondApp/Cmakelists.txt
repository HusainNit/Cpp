cmake_minimum_required(VERSION 3.28)

# this to wake the compiler
project(secondApp)

# make the app portable to other (add c to .exe so it run every where)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")


set(CMAKE_CXX_STANDARD 20) # stander C++20 use
set(CMAKE_CXX_STANDARD_REQUIRED ON) # make sure to use C++20, and fail other wise
set(CMAKE_CXX_EXTENSIONS OFF) # use c++ standerd terms and prevent compiler shortcats


include(FetchContent)

# set(SPDLOG_INSTALL ON) extream and dont need to it in most of the times

FetchContent_Declare(
	spdlog
	GIT_REPOSITORY https://github.com/gabime/spdlog.git
	GIT_TAG v1.17.0
	SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/spdlog"
)

FetchContent_MakeAvailable(spdlog)




# link all the cpp project dynamicaly
file(GLOB_RECURSE MY_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")

# add the files to the IDA looks, use the glope in top - ${PROJECT_NAME} can be better sometimes
add_executable(${PROJECT_NAME} "${MY_SOURCES}")


# Organizing the Files Tree structure
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/src" PREFIX "Source Files" FILES ${MY_SOURCES})



# if we have folder have the resources(player data, art ..etc) we can use the RESOURCES_PATH macro to access these files (to make read and write to file easier)
target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC RESOURCES_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/") # This is useful to get an ASSETS_PATH in your IDE during development but you should comment this if you compile a release version and uncomment the next line
#target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC RESOURCES_PATH="./resources/") # Uncomment this line to setup the ASSETS_PATH macro to the final assets directory when you share the project (production or for others)

# add the files to use with Include
target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include/")

# third party lib include (make sure to use the :: with the alias , and if you have local one add alias to it)
target_link_libraries(${PROJECT_NAME} PRIVATE spdlog::spdlog)



#! ! ! ! ! ! !
#set this to ON instead of OFF to ship the game!
#basically this will change the RESOURCES_PATH to be the local path
#! ! ! ! ! ! !
set(PRODUCTION_BUILD OFF CACHE BOOL "Make this a production build" FORCE)
#delete the out folder after changing if visual studio doesn recognize the change!

#may give problems on linux
if(PRODUCTION_BUILD)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE) #link time optimization
else()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION FALSE)
endif()


# compiler to compile heavy headers one time in bolb if not changed - ${PROJECT_NAME} can be better sometimes
target_precompile_headers(${PROJECT_NAME} PRIVATE
    <vector>
    <string>
    <memory>
    <spdlog/spdlog.h>
)


if(MSVC)  # If using the VS compiler...

    # 1. Rules for YOUR code
    target_compile_options(${PROJECT_NAME} PRIVATE 
        /W4      # Warning Level 4
        /WX      # Treat warnings as errors
        /sdl     # Enable Security Development Lifecycle checks
    )

    # 2. Rules for THIRD PARTY (External) code
    target_compile_options(${PROJECT_NAME} PRIVATE 
        /external:W3            # Treat external headers as Level 3
        /external:anglebrackets # Treat everything in < > as external
    )


    #make sure SIMD optimizations take place
    add_compile_options(/arch:AVX2) 

    # make it multy processer compile
    target_compile_options(${PROJECT_NAME} PRIVATE /MP)



    # Enable faster Debug builds and better "Edit and Continue"
    target_compile_options(${PROJECT_NAME} PRIVATE /Zi) 

    # Better Debug Info Format
    target_compile_options(${PROJECT_NAME} PRIVATE /Zc:inline)
    
    # Improve the "Auto" window in VS - helps the debugger see variables better
    target_compile_options(${PROJECT_NAME} PRIVATE $<$<CONFIG:Debug>:/Od>) # Only for Debug

    
    # allows the compiler to use faster math instructions. It might slightly change very tiny decimal numbers, but for games/apps, the speed boost is massive.
    target_compile_options(${PROJECT_NAME} PRIVATE $<$<CONFIG:Release>:/fp:fast>) # only for Release


    # Enable "Standards Conformance"
    target_compile_options(${PROJECT_NAME} PRIVATE /permissive-)


    # Enable Incremental Linking for faster "Edit and Continue"  - INCREMENTAL
    # Build Speed: "Fast Link"  - DEBUG:FASTLINK
    set_target_properties(${PROJECT_NAME} PROPERTIES LINK_FLAGS_DEBUG "/INCREMENTAL /DEBUG:FASTLINK")


    # define a preprocessor macro when building your target.
	target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC _CRT_SECURE_NO_WARNINGS)

    # The "UTF-8" Fix
    target_compile_options(${PROJECT_NAME} PRIVATE /utf-8)


    # no console in GUI apps
	#set_target_properties("${CMAKE_PROJECT_NAME}" PROPERTIES LINK_FLAGS "/SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup")

endif()
